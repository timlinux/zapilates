{{/* Figure Shortcode — hero-image class opts in to the page-lead style. Usage: {{< figure src="/images/photo.jpg" alt="…" caption="…" class="hero-image" >}} Skips rendering entirely when `src` matches the page's hero image (its front-matter `image`, or the site-wide `heroImage` fallback): that photo is already shown in the page masthead, so we never repeat it in the body. */}} {{ $src := .Get "src" }} {{ $alt := .Get "alt" | default "" }} {{ $caption := .Get "caption" }} {{ $class := .Get "class" | default "" }} {{ $width := .Get "width" }} {{ $hero := or .Page.Params.image .Site.Params.heroImage }} {{ if ne $src $hero }} {{ end }}
